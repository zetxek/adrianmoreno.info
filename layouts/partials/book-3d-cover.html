{{- /* Renders the `.book-3d` scene shared by the grid summary card and the single book page: a CSS box with a front cover (hinges open on hover to reveal an inner page carrying the book's Summary), a spine and a pages edge. The host element must set `position: relative` (or absolute-fill this) and `perspective` (see `.book-cover-frame` / `.book-single-cover-frame` in books.scss). Params (passed as a dict): page - the book Page to render (required) imgClass - class applied to the cover (required) showCta - whether to show the "click to read more" affordance on the inner page (grid cards only; the single page already shows the full text alongside the cover) loading - optional `loading` attribute value for the . "eager" also adds fetchpriority="high" (above-the-fold covers only). width - target 1x display width in px used to pick a resize target; a 2x source is generated too for srcset. Defaults to 400. */ -}} {{- $page := .page -}} {{- $authors := "" -}} {{- if isset $page.Params "book_authors" }}{{ $authors = delimit $page.Params.book_authors ", " }}{{ end -}} {{- $summary := $page.Summary | plainify -}} {{- if $page.Params.cover }} {{- if $summary }} {{ $summary }} {{- if .showCta }} Click to read more → {{- end }} {{- end }} {{- $alt := printf "Cover of %s" $page.Title -}} {{- if $authors }}{{ $alt = printf "Cover of %s by %s" $page.Title $authors }}{{ end -}} {{- $coverRes := resources.Get (strings.TrimPrefix "/" $page.Params.cover) -}} {{- with $coverRes }} {{- $baseWidth := $.width | default 400 -}} {{- $w1x := $baseWidth -}} {{- if lt .Width $baseWidth }}{{ $w1x = .Width }}{{ end -}} {{- $w2xTarget := mul $baseWidth 2 -}} {{- $w2x := $w2xTarget -}} {{- if lt .Width $w2xTarget }}{{ $w2x = .Width }}{{ end -}} {{- $img1x := .Resize (printf "%dx webp q78 Lanczos" $w1x) -}} {{- $img2x := .Resize (printf "%dx webp q78 Lanczos" $w2x) -}} {{- else }} {{- end }} {{- end -}}
